# SQL Interview Preparation - 150 Questions with Detailed Answers

## Sample Database Schema for Practice

```sql
-- Employees table
CREATE TABLE employees (
    employee_id INT PRIMARY KEY,
    first_name VARCHAR(50),
    last_name VARCHAR(50),
    email VARCHAR(100),
    phone VARCHAR(15),
    hire_date DATE,
    job_id VARCHAR(10),
    salary DECIMAL(8,2),
    commission_pct DECIMAL(2,2),
    manager_id INT,
    department_id INT
);

-- Departments table
CREATE TABLE departments (
    department_id INT PRIMARY KEY,
    department_name VARCHAR(30),
    manager_id INT,
    location_id INT
);

-- Jobs table
CREATE TABLE jobs (
    job_id VARCHAR(10) PRIMARY KEY,
    job_title VARCHAR(35),
    min_salary DECIMAL(8,2),
    max_salary DECIMAL(8,2)
);

-- Orders table
CREATE TABLE orders (
    order_id INT PRIMARY KEY,
    customer_id INT,
    order_date DATE,
    total_amount DECIMAL(10,2),
    status VARCHAR(20)
);

-- Order_items table
CREATE TABLE order_items (
    order_item_id INT PRIMARY KEY,
    order_id INT,
    product_id INT,
    quantity INT,
    unit_price DECIMAL(8,2)
);

-- Sample Data
```

### **Q82. Generate employee hierarchy.**
```sql
WITH RECURSIVE employee_hierarchy AS (
    -- Anchor: Top-level managers
    SELECT employee_id, first_name, manager_id, 1 as level
    FROM employees 
    WHERE manager_id IS NULL
    
    UNION ALL
    
    -- Recursive: Subordinates
    SELECT e.employee_id, e.first_name, e.manager_id, eh.level + 1
    FROM employees e
    INNER JOIN employee_hierarchy eh ON e.manager_id = eh.employee_id
)
SELECT * FROM employee_hierarchy ORDER BY level, employee_id;
```

## 🔴 ADVANCED LEVEL (Questions 101-150)

### **Q101. Complex recursive query for organization chart.**
```sql
WITH RECURSIVE org_chart AS (
    SELECT 
        employee_id, 
        first_name, 
        manager_id, 
        salary,
        1 as level,
        CAST(first_name AS CHAR(1000)) as path
    FROM employees 
    WHERE manager_id IS NULL
    
    UNION ALL
    
    SELECT 
        e.employee_id, 
        e.first_name, 
        e.manager_id, 
        e.salary,
        oc.level + 1,
        CONCAT(oc.path, ' -> ', e.first_name)
    FROM employees e
    INNER JOIN org_chart oc ON e.manager_id = oc.employee_id
)
SELECT 
    REPEAT('  ', level - 1) as indent,
    first_name,
    level,
    path,
    salary
FROM org_chart 
ORDER BY path;
```

### **Q102. Advanced window functions for salary analysis.**
```sql

    first_name,

    -- Ranking functions
```

```sql
            ELSE total_orders
        END as orders_per_month
    FROM customer_metrics
```

```sql
        -- Team salary statistics for managers
        CASE WHEN hm.direct_reports > 0 THEN
            (SELECT AVG(salary) FROM employees WHERE manager_id = hm.employee_id)
        END as team_avg_salary,
        
        CASE WHEN hm.direct_reports > 0 THEN
            (SELECT MAX(salary) FROM employees WHERE manager_id = hm.employee_id)
        END as team_max_salary
        
    FROM hierarchy_metrics hm
)
SELECT 
    REPEAT('  ', level - 1) as indentation,
    CONCAT(first_name, ' ', last_name) as full_name,
    level as org_level,
    direct_reports,
    total_subordinates,
    departments_managed,
    ROUND(salary, 0) as salary,
    
    -- Salary analysis
    CASE 
        WHEN manager_salary IS NOT NULL THEN 
            ROUND((salary - manager_salary) / manager_salary * 100, 1)
        ELSE NULL
    END as salary_vs_manager_pct,
    
    ROUND((salary - peer_avg_salary) / peer_avg_salary * 100, 1) as salary_vs_peers_pct,
    
    -- Management effectiveness
    CASE 
        WHEN direct_reports = 0 THEN 'Individual Contributor'
        WHEN direct_reports <= 3 THEN 'Team Leader'
        WHEN direct_reports <= 8 THEN 'Manager'
        ELSE 'Senior Manager'
    END as management_level,
    
    -- Compensation fairness analysis
    CASE 
        WHEN team_avg_salary IS NOT NULL AND salary < team_avg_salary THEN 'Underpaid vs Team'
        WHEN manager_salary IS NOT NULL AND salary >= manager_salary THEN 'Overpaid vs Manager'
        ELSE 'Appropriately Paid'
    END as compensation_assessment,

    CASE 
        WHEN level = 1 THEN 'C-Level/VP'
        WHEN level = 2 AND direct_reports > 0 THEN 'Director'
        WHEN level = 3 AND direct_reports > 0 THEN 'Manager'
        WHEN level >= 4 THEN 'Senior/Lead'
        ELSE 'Staff'
    END as suggested_title,

FROM salary_comparisons
```

```sql
        WHEN org_level = 2 AND direct_reports > 0 THEN 'Senior Management'
        WHEN org_level = 3 AND direct_reports > 0 THEN 'Middle Management'
        WHEN direct_reports > 0 THEN 'Team Lead'
        WHEN years_employed > 5 AND company_salary_percentile > 0.7 THEN 'Senior IC'
        ELSE 'Staff Level'
    END as career_track,
    
    -- Context for decision making
    CONCAT(
        'Reports to: ', (SELECT CONCAT(first_name, ' ', last_name) FROM employees WHERE employee_id = ro.manager_id),
        ' | Team Size: ', direct_reports,
        ' | Dept Rank: #', dept_salary_rank,
        ' | Tenure: ', years_employed, ' years'
    ) as context_summary
    
FROM risk_opportunity ro
ORDER BY 
    department_id, 
    org_level, 
    CASE 
        WHEN career_status IN ('High Potential', 'Promotion Overdue') THEN 1
        WHEN career_status IN ('Career Plateau Risk', 'Retention Risk') THEN 2
        ELSE 3
    END,
    salary DESC;
```

### **Q143. Find salary outliers using IQR method.**
```sql
WITH salary_quartiles AS (
    SELECT 
        PERCENTILE_CONT(0.25) WITHIN GROUP (ORDER BY salary) as Q1,
        PERCENTILE_CONT(0.75) WITHIN GROUP (ORDER BY salary) as Q3
    FROM employees
),
outlier_bounds AS (
    SELECT 
        Q1,
        Q3,
        Q3 - Q1 as IQR,
        Q1 - 1.5 * (Q3 - Q1) as lower_bound,
        Q3 + 1.5 * (Q3 - Q1) as upper_bound
    FROM salary_quartiles
)
SELECT e.first_name, e.salary,
       CASE 
           WHEN e.salary < ob.lower_bound THEN 'Low Outlier'
           WHEN e.salary > ob.upper_bound THEN 'High Outlier'
           ELSE 'Normal'
       END as outlier_status
FROM employees e
CROSS JOIN outlier_bounds ob
WHERE e.salary < ob.lower_bound OR e.salary > ob.upper_bound
ORDER BY e.salary;
```

### **Q146. Find the longest gap between consecutive hire dates.**

```sql
    FROM employees
    ORDER BY hire_date
```

### **Q150. Master query - Complete employee analytics dashboard.**
```sql
WITH employee_stats AS (
    SELECT 
        employee_id, first_name, last_name, department_id, salary, hire_date,
        DATEDIFF(CURRENT_DATE, hire_date) as days_employed,
        RANK() OVER (ORDER BY salary DESC) as company_rank,
        RANK() OVER (PARTITION BY department_id ORDER BY salary DESC) as dept_rank,
        COUNT(*) OVER (PARTITION BY department_id) as dept_size,
        AVG(salary) OVER (PARTITION BY department_id) as dept_avg_salary,
        (SELECT COUNT(*) FROM employees WHERE manager_id = e.employee_id) as direct_reports
    FROM employees e
),
performance_tiers AS (
    SELECT *,
        CASE 
            WHEN company_rank <= (SELECT COUNT(*) * 0.1 FROM employees) THEN 'Top 10%'
            WHEN company_rank <= (SELECT COUNT(*) * 0.25 FROM employees) THEN 'Top 25%'
            WHEN company_rank <= (SELECT COUNT(*) * 0.5 FROM employees) THEN 'Top 50%'
            ELSE 'Bottom 50%'
        END as performance_tier,
        CASE 
            WHEN days_employed >= 1825 THEN 'Veteran (5+ years)'  -- 5 years
            WHEN days_employed >= 1095 THEN 'Experienced (3-5 years)'  -- 3 years
            WHEN days_employed >= 365 THEN 'Intermediate (1-3 years)'
            ELSE 'New (< 1 year)'
        END as tenure_category
    FROM employee_stats
)
SELECT 
    CONCAT(first_name, ' ', last_name) as employee_name,
    department_id,
    salary,
    performance_tier,
    tenure_category,
    CONCAT('#', company_rank, ' of ', (SELECT COUNT(*) FROM employees)) as company_ranking,
    CONCAT('#', dept_rank, ' of ', dept_size) as department_ranking,
    ROUND(salary * 100.0 / dept_avg_salary, 1) as salary_vs_dept_avg_pct,
    direct_reports,
    ROUND(days_employed / 365.25, 1) as years_employed,
    
    -- Key insights
    CASE 
        WHEN performance_tier = 'Top 10%' AND tenure_category LIKE 'New%' THEN 'HIGH POTENTIAL'
        WHEN performance_tier IN ('Bottom 50%') AND tenure_category LIKE 'Veteran%' THEN 'RETENTION RISK'
        WHEN direct_reports > 0 AND performance_tier = 'Top 25%' THEN 'STRONG LEADER'
        WHEN salary > dept_avg_salary * 1.5 AND direct_reports = 0 THEN 'SENIOR IC'
        ELSE 'STANDARD'
    END as employee_classification,
    
    -- Action items
    CASE 
        WHEN performance_tier = 'Top 10%' THEN 'Consider for promotion/leadership roles'
        WHEN performance_tier = 'Bottom 50%' AND days_employed > 1095 THEN 'Performance review recommended'  
        WHEN salary < dept_avg_salary * 0.8 AND days_employed > 730 THEN 'Salary review recommended'
        ELSE 'Continue monitoring'
    END as recommended_action
    
FROM performance_tiers
ORDER BY department_id, company_rank;
```
